"""
genesis_inverse_iterate.py   (3+1 NESTED version)
==================================================

Four-level nested oscillator: 1 container + 3 internal scales.

Stephen's flat-vs-nested experiment plus the vote decomposition:

    Level 0 (CONTAINER):    the observable.  We read the nonce from here.
    Level 1 (internal):     coarse correction, crossing D -> sqrt(D)
                            contributes (D - sqrt(D)) / f(K,0) of the total
    Level 2 (internal):     refined correction, crossing sqrt(D) -> 1
                            contributes (sqrt(D) - 1) / f(K,0), attenuated
                            from L1 by factor 1/sqrt(D)
    Level 3 (closure):      contributes ~0 when D=3 closure holds

In the GFD physics this produces the 2/17 shift because D=3 makes the
third-level correction vanish identically.  For SHA256 we don't have
D=3 algebraic closure -- but we import the STRUCTURE:

    * 4 oscillators on the same circuit
    * coupling strength attenuates by sqrt(3) each deeper level
    * bottom-up update sequence each step (L3 -> L2 -> L1 -> L0)
    * consensus lock across all 4 before clamping a wire
    * answer READ FROM L0 only

The four-way consensus is strictly more conservative than the three-way
we tried before.  A wire only locks when:
    mu_L0 < tol AND mu_L1 < tol AND mu_L2 < tol AND mu_L3 < tol   (lock to 0)
    mu_L0 > 1-tol AND same for L1,L2,L3                            (lock to 1)
If L3 (the closure layer) disagrees, we don't lock -- that's the fractal
telling us the bit hasn't "closed" yet.

Expected: the 0.2% wrong-lock rate from single-osc should approach 0.
The question is whether enough wires pass four-way consensus to maintain
a cascade, or whether L3 (which has the weakest coupling to the
boundary) holds everything hostage by never committing.
"""

class GateCircuit:
    def __init__(self):
        self.init_values = []
        self.clamped     = []
        self.xor_gates = []
        self.and_gates = []
        self.or_gates  = []
        self.not_gates = []

    def add_wire(self, init=0.5, clamped=False):
        idx = len(self.init_values)
        self.init_values.append(float(init))
        self.clamped.append(bool(clamped))
        return idx

    def add_word_const(self, value, width=32):
        return [self.add_wire(init=(value >> (width - 1 - i)) & 1, clamped=True)
                for i in range(width)]

    def add_word_free(self, width=32):
        return [self.add_wire() for _ in range(width)]

    def xor(self, a, b):
        c = self.add_wire(); self.xor_gates.append((a, b, c)); return c
    def and_(self, a, b):
        c = self.add_wire(); self.and_gates.append((a, b, c)); return c
    def or_(self, a, b):
        c = self.add_wire(); self.or_gates.append((a, b, c)); return c
    def not_(self, a):
        c = self.add_wire(); self.not_gates.append((a, c)); return c

    def xor32(self, a, b): return [self.xor(a[i], b[i]) for i in range(32)]
    def and32(self, a, b): return [self.and_(a[i], b[i]) for i in range(32)]
    def not32(self, a):    return [self.not_(x) for x in a]

    def rotr32(self, a, n):
        return a[-n:] + a[:-n] if n else a[:]

    def shr32(self, a, n):
        if n == 0: return a[:]
        zero = self.add_wire(init=0.0, clamped=True)
        return [zero] * n + a[:-n]

    def add32(self, a, b):
        s = [None] * 32
        carry = None
        for i in range(31, -1, -1):
            if carry is None:
                s[i]  = self.xor(a[i], b[i])
                carry = self.and_(a[i], b[i])
            else:
                ab_xor          = self.xor(a[i], b[i])
                s[i]            = self.xor(ab_xor, carry)
                a_and_b         = self.and_(a[i], b[i])
                carry_and_prop  = self.and_(carry, ab_xor)
                carry           = self.or_(a_and_b, carry_and_prop)
        return s

    def Ch(self, e, f, g):
        ef = self.and32(e, f); not_e = self.not32(e); ne_g = self.and32(not_e, g)
        return self.xor32(ef, ne_g)

    def Maj(self, a, b, c):
        ab = self.and32(a, b); ac = self.and32(a, c); bc = self.and32(b, c)
        return self.xor32(self.xor32(ab, ac), bc)

    def Sigma0(self, x):
        return self.xor32(self.xor32(self.rotr32(x, 2), self.rotr32(x, 13)),
                          self.rotr32(x, 22))

    def Sigma1(self, x):
        return self.xor32(self.xor32(self.rotr32(x, 6), self.rotr32(x, 11)),
                          self.rotr32(x, 25))

    def sigma0(self, x):
        return self.xor32(self.xor32(self.rotr32(x, 7), self.rotr32(x, 18)),
                          self.shr32(x, 3))

    def sigma1(self, x):
        return self.xor32(self.xor32(self.rotr32(x, 17), self.rotr32(x, 19)),
                          self.shr32(x, 10))

    def sha256_block(self, W_in, H_in):
        W = list(W_in)
        for t in range(16, 64):
            W.append(self.add32(
                self.add32(self.sigma1(W[t-2]), W[t-7]),
                self.add32(self.sigma0(W[t-15]), W[t-16])
            ))
        a, b, c, d, e, f, g, h = [H_in[i] for i in range(8)]
        for t in range(64):
            K_t = self.add_word_const(K[t])
            T1 = self.add32(
                    self.add32(
                        self.add32(h, self.Sigma1(e)),
                        self.add32(self.Ch(e, f, g), K_t)
                    ),
                    W[t]
                 )
            T2 = self.add32(self.Sigma0(a), self.Maj(a, b, c))
            h = g; g = f; f = e
            e = self.add32(d, T1)
            d = c; c = b; b = a
            a = self.add32(T1, T2)
        return [self.add32(H_in[0], a),
                self.add32(H_in[1], b),
                self.add32(H_in[2], c),
                self.add32(H_in[3], d),
                self.add32(H_in[4], e),
                self.add32(H_in[5], f),
                self.add32(H_in[6], g),
                self.add32(H_in[7], h)]


def bytes_to_words_MSB(data):
    return list(struct.unpack(">16I", data))


def build_full_genesis_circuit():
    cc = GateCircuit()
    header = build_header(0)

    blk1 = bytes_to_words_MSB(header[:64])
    W1 = [cc.add_word_const(word) for word in blk1]

    W2 = []
    for wi in range(3):
        w_bytes = header[64 + 4*wi : 64 + 4*wi + 4]
        W2.append(cc.add_word_const(int.from_bytes(w_bytes, "big")))
    nonce_wires = cc.add_word_free()
    W2.append(nonce_wires)
    W2.append(cc.add_word_const(0x80000000))
    for _ in range(10):
        W2.append(cc.add_word_const(0))
    W2.append(cc.add_word_const(640))

    H1 = [cc.add_word_const(H_INIT[i]) for i in range(8)]
    H1_after_blk1 = cc.sha256_block(W1, H1)
    H1_after_blk2 = cc.sha256_block(W2, H1_after_blk1)

    W3 = list(H1_after_blk2)
    W3.append(cc.add_word_const(0x80000000))
    for _ in range(6):
        W3.append(cc.add_word_const(0))
    W3.append(cc.add_word_const(256))

    H2 = [cc.add_word_const(H_INIT[i]) for i in range(8)]
    H2_after = cc.sha256_block(W3, H2)

    out_wires = []
    for word_wires in H2_after:
        for w in word_wires:
            out_wires.append(w)
    for i in range(224, 256):
        cc.clamped[out_wires[i]] = True
        cc.init_values[out_wires[i]] = 0.0

    return cc, nonce_wires, out_wires


def circuit_to_arrays(cc):
    return {
        "mu_init":    np.array(cc.init_values, dtype=np.float64),
        "clamped":    np.array(cc.clamped, dtype=bool),
        "xor_gates":  np.array(cc.xor_gates, dtype=np.int32),
        "and_gates":  np.array(cc.and_gates, dtype=np.int32),
        "or_gates":   np.array(cc.or_gates, dtype=np.int32),
        "not_gates":  np.array(cc.not_gates, dtype=np.int32),
    }


def forward_evaluate(arrays, nonce_value_int, nonce_wires):
    mu = arrays["mu_init"].copy()
    nonce_word = int_to_circuit_word(nonce_value_int)
    for i, w in enumerate(nonce_wires):
        mu[w] = float((nonce_word >> (31 - i)) & 1)

    gate_for_output = {}
    for row in arrays["xor_gates"]:
        gate_for_output[int(row[2])] = ("xor", int(row[0]), int(row[1]))
    for row in arrays["and_gates"]:
        gate_for_output[int(row[2])] = ("and", int(row[0]), int(row[1]))
    for row in arrays["or_gates"]:
        gate_for_output[int(row[2])] = ("or",  int(row[0]), int(row[1]))
    for row in arrays["not_gates"]:
        gate_for_output[int(row[1])] = ("not", int(row[0]), None)

    for c in range(mu.shape[0]):
        g = gate_for_output.get(c)
        if g is None: continue
        op, a, b = g
        if   op == "xor": mu[c] = mu[a] + mu[b] - 2 * mu[a] * mu[b]
        elif op == "and": mu[c] = mu[a] * mu[b]
        elif op == "or":  mu[c] = mu[a] + mu[b] - mu[a] * mu[b]
        elif op == "not": mu[c] = 1.0 - mu[a]
    return mu


def bits_from_mu(mu, wires):
    v = 0
    for i, w in enumerate(wires):
        if mu[w] > 0.5:
            v |= (1 << (len(wires) - 1 - i))
    return v


def out_wires_to_display_hex(mu, out_wires):
    out_bits = [int(mu[w] > 0.5) for w in out_wires]
    raw = bytearray(32)
    for i in range(256):
        raw[i // 8] |= out_bits[i] << (7 - (i % 8))
    return bytes(raw[::-1]).hex(), bytes(raw).hex()


def verify_circuit(arrays, nonce_wires, out_wires):
    mu = forward_evaluate(arrays, KNOWN_NONCE, nonce_wires)
    h_display, _ = out_wires_to_display_hex(mu, out_wires)
    ok = (h_display == KNOWN_HASH_DISPLAY)
    print(f"  Forward sanity: match = {ok}   (hash {h_display})")
    return ok

def gate_gradient(arrays, mu):
    """Standard gate-residual gradient on a single mu vector."""
    XOR = arrays["xor_gates"]
    AND = arrays["and_gates"]
    OR  = arrays["or_gates"]
    NOT = arrays["not_gates"]

    grad = np.zeros_like(mu)
    total_loss = 0.0

    if len(XOR):
        a, b, c = XOR[:, 0], XOR[:, 1], XOR[:, 2]
        ma, mb, mc = mu[a], mu[b], mu[c]
        r = ma + mb - 2 * ma * mb - mc
        np.add.at(grad, a, 2 * r * (1 - 2 * mb))
        np.add.at(grad, b, 2 * r * (1 - 2 * ma))
        np.add.at(grad, c, -2 * r)
        total_loss += (r * r).sum()

    if len(AND):
        a, b, c = AND[:, 0], AND[:, 1], AND[:, 2]
        ma, mb, mc = mu[a], mu[b], mu[c]
        r = ma * mb - mc
        np.add.at(grad, a, 2 * r * mb)
        np.add.at(grad, b, 2 * r * ma)
        np.add.at(grad, c, -2 * r)
        total_loss += (r * r).sum()

    if len(OR):
        a, b, c = OR[:, 0], OR[:, 1], OR[:, 2]
        ma, mb, mc = mu[a], mu[b], mu[c]
        r = ma + mb - ma * mb - mc
        np.add.at(grad, a, 2 * r * (1 - mb))
        np.add.at(grad, b, 2 * r * (1 - ma))
        np.add.at(grad, c, -2 * r)
        total_loss += (r * r).sum()

    if len(NOT):
        a, c = NOT[:, 0], NOT[:, 1]
        ma, mc = mu[a], mu[c]
        r = 1 - ma - mc
        np.add.at(grad, a, -2 * r)
        np.add.at(grad, c, -2 * r)
        total_loss += (r * r).sum()

    return grad, total_loss


def relax_nested_3plus1(arrays, n_steps, lr, coupling_base,
                         initial_mus, seeds):
    """4-level nested relaxation.

    Levels:
        L0 = CONTAINER (observable)
        L1, L2, L3 = internal scales

    Couplings (attenuated by sqrt(3) per level crossing):
        L0<->L1:  coupling_base
        L1<->L2:  coupling_base / sqrt(3)
        L2<->L3:  coupling_base / 3

    Update order each step: bottom-up (L3, L2, L1, L0).
    The leaf (L3) updates first -- pure local dynamics, no children
    to wait on.  Each level above waits for its child to have already
    moved this step before incorporating it.
    """
    K = 4
    clamped = arrays["clamped"]
    free = ~clamped

    # Coupling strengths between adjacent levels, attenuated by sqrt(3)
    # couplings[k] = L_k <-> L_{k+1}
    couplings = [
        coupling_base,                # L0-L1
        coupling_base / SQRT3,        # L1-L2
        coupling_base / 3.0,          # L2-L3
    ]

    # Initialise four mu vectors
    mus = []
    for k in range(K):
        if initial_mus[k] is not None:
            mu = initial_mus[k].copy()
        else:
            rng = np.random.default_rng(seeds[k])
            mu = arrays["mu_init"].copy()
            mu[free] = 0.5 + 0.01 * rng.standard_normal(free.sum())
            mu[free] = np.clip(mu[free], 0.02, 0.98)
        mu[clamped] = arrays["mu_init"][clamped]
        mus.append(mu)

    losses = [np.zeros(n_steps) for _ in range(K)]
    # Per-step history for detecting lock order
    level_changes = [np.zeros(n_steps) for _ in range(K)]

    for step in range(n_steps):
        # --- Update BOTTOM-UP: L3, L2, L1, L0 -------------------------
        # Each level uses the already-updated-this-step state of its
        # child.  This produces the bottom-up info flow.
        for k in range(K - 1, -1, -1):
            mu_k_prev = mus[k].copy()

            grad_k, loss_k = gate_gradient(arrays, mus[k])
            losses[k][step] = loss_k

            # Pull toward parent (if any)
            if k > 0:
                coupling = couplings[k - 1]
                grad_k += coupling * (mus[k] - mus[k - 1])

            # Pull toward child (if any) -- weaker (upward info flow)
            if k < K - 1:
                coupling = couplings[k]
                grad_k += 0.3 * coupling * (mus[k] - mus[k + 1])

            grad_k[clamped] = 0.0
            mus[k] = mus[k] - lr * grad_k
            np.clip(mus[k], 0.0, 1.0, out=mus[k])
            mus[k][clamped] = arrays["mu_init"][clamped]

            # Measure how much this level moved this step
            level_changes[k][step] = float(
                np.abs(mus[k][free] - mu_k_prev[free]).mean()
            )

    return mus, losses, level_changes


# =====================================================================
#   LOCK-AND-ITERATE with 3+1 NESTED ensemble
# =====================================================================

def lock_and_iterate_3plus1(arrays, nonce_wires, out_wires,
                             coupling_base=COUPLING_BASE,
                             lr=LEARNING_RATE,
                             lock_tol=LOCK_TOL,
                             max_rounds=MAX_ROUNDS,
                             steps_per_round=STEPS_PER_ROUND,
                             truth_bits=None,
                             initial_clamped=None):

    K = 4
    print(f"3+1 nested lock-and-iterate:")
    print(f"  levels: L0 (container, readout), L1, L2, L3 (closure)")
    print(f"  couplings: L0<->L1={coupling_base}, "
          f"L1<->L2={coupling_base/SQRT3:.4f}, "
          f"L2<->L3={coupling_base/3:.4f}  (ratio 1/sqrt(3))")
    print(f"  lr = {lr}  lock_tol = {lock_tol}  "
          f"unanimous consensus across all 4 levels")
    print(f"  reading nonce from L0 (container)")
    print()
    header = (f"{'Rnd':>3}  {'lock':>6}  {'rejct':>5}  "
              f"{'clamped':>8} ({'%':>4})  {'nonce':>5}  "
              f"{'L0':>9}  {'L1':>9}  {'L2':>9}  {'L3':>9}  "
              f"{'corr':>7}  {'wrng':>5}  {'acc%':>5}  {'L0 nonce':>10}")
    print(header)
    print("-" * len(header))

    n_wires = arrays["mu_init"].shape[0]
    mus = [None] * K

    history = {
        "round": [], "new_locks": [], "rejected": [],
        "total_clamped": [], "nonce_locked": [],
        "loss_per_k": [], "nonce_mu_L0": [],
        "correct_locks": [], "wrong_locks": [], "accuracy": [],
        "current_nonce_L0": [],
        "losses_per_k_per_round": [],
        "level_changes_per_round": [],
    }

    for rnd in range(max_rounds):
        seeds = [rnd * K + k + 1 for k in range(K)]
        mus, losses_per_k, level_changes = relax_nested_3plus1(
            arrays, n_steps=steps_per_round,
            lr=lr, coupling_base=coupling_base,
            initial_mus=mus, seeds=seeds)

        # Unanimous consensus across all 4 levels
        mu_stack = np.stack(mus, axis=0)        # (4, n_wires)
        free = ~arrays["clamped"]

        all_low  = np.all(mu_stack < lock_tol, axis=0)
        all_high = np.all(mu_stack > 1.0 - lock_tol, axis=0)
        close_0 = all_low & free
        close_1 = all_high & free
        n_new = int(close_0.sum() + close_1.sum())

        any_low  = np.any(mu_stack < lock_tol, axis=0)
        any_high = np.any(mu_stack > 1.0 - lock_tol, axis=0)
        any_threshold = (any_low | any_high) & free
        blocked = any_threshold & ~(close_0 | close_1)
        n_rejected = int(blocked.sum())

        arrays["clamped"][close_0] = True
        arrays["mu_init"][close_0] = 0.0
        arrays["clamped"][close_1] = True
        arrays["mu_init"][close_1] = 1.0
        for k in range(K):
            mus[k][close_0] = 0.0
            mus[k][close_1] = 1.0

        # Audit
        lock_added = arrays["clamped"] & (~initial_clamped)
        locked_idx = np.where(lock_added)[0]
        locked_vals = arrays["mu_init"][locked_idx].round().astype(int)
        true_vals = truth_bits[locked_idx]
        correct_mask = (locked_vals == true_vals)
        n_correct = int(correct_mask.sum())
        n_wrong = int(lock_added.sum()) - n_correct
        acc = 100.0 * n_correct / max(int(lock_added.sum()), 1)

        total_clamped = int(arrays["clamped"].sum())
        locked_pct = 100.0 * total_clamped / n_wires

        # Read nonce from L0 only (the container)
        nonce_arr = np.array(nonce_wires)
        nonce_mu_L0 = mus[0][nonce_arr].copy()
        nonce_locked = int(((nonce_mu_L0 < lock_tol) |
                            (nonce_mu_L0 > 1.0 - lock_tol)).sum())
        nonce_word = bits_from_mu(mus[0], nonce_wires)
        current_nonce_L0 = circuit_word_to_int(nonce_word)

        final_losses = [losses_per_k[k][-1] for k in range(K)]

        history["round"].append(rnd)
        history["new_locks"].append(n_new)
        history["rejected"].append(n_rejected)
        history["total_clamped"].append(total_clamped)
        history["nonce_locked"].append(nonce_locked)
        history["loss_per_k"].append(final_losses)
        history["nonce_mu_L0"].append(nonce_mu_L0)
        history["correct_locks"].append(n_correct)
        history["wrong_locks"].append(n_wrong)
        history["accuracy"].append(acc)
        history["current_nonce_L0"].append(current_nonce_L0)
        history["losses_per_k_per_round"].append(
            [L.copy() for L in losses_per_k])
        history["level_changes_per_round"].append(
            [c.copy() for c in level_changes])

        print(f"{rnd:>3d}  {n_new:>6,}  {n_rejected:>5,}  "
              f"{total_clamped:>8,} ({locked_pct:>4.1f}%)  "
              f"{nonce_locked:>3d}/32  "
              f"{final_losses[0]:>9.2e}  {final_losses[1]:>9.2e}  "
              f"{final_losses[2]:>9.2e}  {final_losses[3]:>9.2e}  "
              f"{n_correct:>7,}  {n_wrong:>5,}  {acc:>5.2f}  "
              f"{current_nonce_L0:>10d}")

        if n_new == 0:
            print(f"\nNo new consensus locks in round {rnd}; terminating.")
            break
        if nonce_locked == 32:
            print(f"\nAll 32 L0 nonce bits locked with 4-way consensus!")
            break

    return mus, history
